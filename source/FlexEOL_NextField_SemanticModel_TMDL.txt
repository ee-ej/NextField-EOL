//   workbook (33 sheets, 193-col asset DB, 9,070 equipment rows) and the exact
//   9-criteria weighted scoring model from the 'Criteria' tab.
//
// SOURCE OF TRUTH (verified from the .xlsx, 2026-09-03)
//   Fact_AssetScore      <- 'Output- Asset Scores'        (header row 16, 22 cols)
//   Fact_SiteRisk        <- 'Output- Site Risk Scores' +  'Output- Site Scores'
//   Fact_CapitalForecast <- 'Output- Budget Forecasting'  (Year 1..10)
//   Dim_Asset            <- 'Entry- Eq Database'          (header row 21, 193 cols)
//   Dim_Site             <- 'Entry- Site Data'            (header row 20, 31 cols)
//   Dim_ScoringWeights   <- 'Criteria'                    (rows 16-24)
//   Dim_RiskWeights      <- 'Risk Eval Weighting'

//   Criticality 100 | Operating Issues 88.888..| Redundancy 77.777..
//   Service & Support 66.666.. | Age 55.555.. | Site Business Priority 44.444..
//   Efficiency 33.333.. | Maintenance Cost 22.222.. | Loading 11.111..
//
// COST MODEL (from 'Costs and Scoring' tab)
//   ROM scaling uses "Rule of Six-Tenths" family but the workbook's actual
//   "Exponent Applied for Size Variance" = 0.8 for ALL equipment classes.
//   CostB = CostA * (SizeB / SizeA) ^ 0.8
//
// NOTE
//   Mode is set to import for the PoC seed from the workbook. Switch tables to
//   Direct Lake once the gold Lakehouse tables exist. Replace M partition

table Fact_AssetScore
	lineageTag: fact-assetscore-0001

	measure 'Overall Score (avg)' = AVERAGE(Fact_AssetScore[OverallScore])
		formatString: 0.0
		displayFolder: 1 Scores

	/// Weighted Overall Score rebuilt from the 9 sub-scores using the tunable
	/// weights in Dim_ScoringWeights. Use this to prove the workbook math is
	/// reproducible; swap to the precomputed [OverallScore] for performance.
	measure 'Overall Score (weighted, dynamic)' = ```
			VAR w = Dim_ScoringWeights
			VAR wCrit = MAXX(FILTER(w, w[Criteria] = "Criticality"), w[Weighting])
			VAR wOps  = MAXX(FILTER(w, w[Criteria] = "Operating Issues"), w[Weighting])
			VAR wRed  = MAXX(FILTER(w, w[Criteria] = "Redundancy"), w[Weighting])
			VAR wSvc  = MAXX(FILTER(w, w[Criteria] = "Service and Support"), w[Weighting])
			VAR wAge  = MAXX(FILTER(w, w[Criteria] = "Age"), w[Weighting])
			VAR wSbp  = MAXX(FILTER(w, w[Criteria] = "Site Business Priority"), w[Weighting])
			VAR wEff  = MAXX(FILTER(w, w[Criteria] = "Efficiency"), w[Weighting])
			VAR wMnt  = MAXX(FILTER(w, w[Criteria] = "Maintenance Cost"), w[Weighting])
			VAR wLod  = MAXX(FILTER(w, w[Criteria] = "Loading / Equipment Taxing"), w[Weighting])
			VAR wTot  = wCrit + wOps + wRed + wSvc + wAge + wSbp + wEff + wMnt + wLod
			RETURN
			DIVIDE(
			    SUMX(
			        Fact_AssetScore,
			          Fact_AssetScore[EqCriticality]        * wCrit
			        + Fact_AssetScore[EqOperatingIssues]    * wOps
			        + Fact_AssetScore[EqRedundancy]         * wRed
			        + Fact_AssetScore[EqServiceAndSupport]  * wSvc
			        + Fact_AssetScore[EqAge]                * wAge
			        + Fact_AssetScore[SiteBusinessPriority] * wSbp
			        + Fact_AssetScore[EqEfficiency]         * wEff
			        + Fact_AssetScore[EqMainCost]           * wMnt
			        + Fact_AssetScore[EqLoading]            * wLod
			    ),
			    COUNTROWS(Fact_AssetScore) * wTot
			)
			```
		formatString: 0.0
		displayFolder: 1 Scores

	measure 'Asset Count' = DISTINCTCOUNT(Fact_AssetScore[AssetId])
		formatString: #,0
		displayFolder: 0 Counts

	/// Replacement-threshold parameter: workbook flags score >= 50 as replace-now.
	measure 'Assets At/Over Threshold' = ```
			VAR t = [Replacement Threshold Value]
			RETURN CALCULATE([Asset Count], Fact_AssetScore[OverallScore] >= t)
			```
		formatString: #,0

	measure 'ROM Cost in Budget Year ($)' = SUM(Fact_AssetScore[ROMCostInBudgetYear])
